  const isPhone = v => /^[+\d][\d\s()\-]{6,18}$/.test(v);

  function mark(fieldWrap, bad) { fieldWrap.classList.toggle('error', bad); return bad; }

  form.addEventListener('submit', e => {
    e.preventDefault();

    const nameV = fName.value.trim();
    const phoneV = fPhone.value.trim();
    const emailV = fEmail.value.trim();
    const msgV = fMsg.value.trim();

    // Валидация
    let bad = false;
    bad = mark(document.getElementById('fieldName'), nameV.length < 2) || bad;

    const phoneBad = phoneV ? !isPhone(phoneV) : !emailV;
    const emailBad = emailV ? !isEmail(emailV) : !phoneV;
    bad = mark(document.getElementById('fieldPhone'), phoneBad) || bad;
    bad = mark(document.getElementById('fieldEmail'), emailBad) || bad;

    bad = mark(document.getElementById('fieldMsg'), msgV.length < 10) || bad;

    if (bad) { showToast('Проверьте выделенные поля формы 🙂'); return; }

    // Отправка в Netlify (форма name="zayavka", data-netlify="true")
    const params = new URLSearchParams();
    params.append('form-name', 'zayavka');
    params.append('name', nameV);
    params.append('phone', phoneV);
    params.append('email', emailV);
    params.append('comment', msgV);

    fetch('/', {
      method: 'POST',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
      body: params.toString()
    })
    .then(res => {
      if (!res.ok) throw new Error('Netlify error');
      showToast('✅ Заявка отправлена! Мы свяжемся с вами в ближайшее время.');
      form.reset();
    })
    .catch(() => {
      // Резервный вариант: письмо на cvr09@yandex.ru с пометкой
      const subject = encodeURIComponent('Заявка с сайта Точка добра');
      const body = encodeURIComponent(
        'Имя: ' + nameV + '\n' +
        'Телефон: ' + phoneV + '\n' +
        'E-mail: ' + emailV + '\n\n' +
        'Комментарий:\n' + msgV
      );
      window.location.href = 'mailto:cvr09@yandex.ru?subject=' + subject + '&body=' + body;
      showToast('Открываем почтовую программу для отправки заявки…');
      setTimeout(() => form.reset(), 1000);
    });
  });

  [fName, fPhone, fEmail, fMsg].forEach(inp =>
    inp.addEventListener('input', () => inp.closest('.field').classList.remove('error'))
  );

});
</script>
</body>
</html>
